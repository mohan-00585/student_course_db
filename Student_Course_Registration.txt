create database student_course_registration;
use student_course_registration;

create table students (
    student_id int primary key,
    student_name varchar(50) not null,
    gender varchar(10) not null,
    date_of_birth date not null,
    phone varchar(15) not null unique,
    email varchar(100) unique,
    city varchar(50) not null,
    check (gender in ('Male', 'Female'))
);

create table courses (
    course_id int primary key,
    course_name varchar(100) not null unique,
    category varchar(50) not null,
    duration_days int not null check (duration_days > 0),
    course_fee decimal(10,2) not null check (course_fee >= 0)
);

create table enrollments (
    enrollment_id int primary key,
    student_id int not null,
    course_id int not null,
    enrollment_date date not null,
    fee_paid decimal(10,2) not null default 0 check (fee_paid >= 0),
    score decimal(5,2) check (score between 0 and 100),
    enrollment_status varchar(15) not null default 'Enrolled',
    foreign key (student_id) references students(student_id),
    foreign key (course_id) references courses(course_id),
    unique (student_id, course_id),
    check (enrollment_status in ('Enrolled', 'Completed', 'Dropped')),
    check (enrollment_status <> 'Completed' or score is not null)
);

select * from students;
select * from courses;
select * from enrollments;

select student_name, gender, city from students where city in ('Theni', 'Chennai');

select student_name, phone, email from students where student_name like 'A%' and email is not null;

select course_name, category, course_fee from courses where course_fee between 5000 and 10000 
order by course_fee desc;

select course_name, duration_days, course_fee from courses where duration_days > 30 
order by duration_days;

select enrollment_id, enrollment_date, fee_paid, enrollment_status from enrollments where 
enrollment_status = 'Completed' order by enrollment_date desc;

select city, count(*) as total_students from students group by city order by total_students desc;

select category, avg(course_fee) as avg_fee from courses group by category;

select enrollment_status, count(*) as total_enrollments, sum(fee_paid) as total_fee from 
enrollments group by enrollment_status;

select s.student_name, s.city, e.enrollment_date, e.fee_paid from students s inner join 
enrollments e on s.student_id = e.student_id;

select c.course_name, c.category, e.score, e.enrollment_status from courses c inner join 
enrollments e on c.course_id = e.course_id where e.score >= 80;

select s.student_name, s.gender, c.course_name, c.category, e.enrollment_date, e.fee_paid, e.score 
from students s join enrollments e on s.student_id = e.student_id join courses c on e.course_id = c.course_id;

select s.student_name, e.enrollment_status, e.score from students s left join enrollments e
 on s.student_id = e.student_id;

select c.course_name, count(e.enrollment_id) as total_enrollments from courses c left join
 enrollments e on c.course_id = e.course_id group by c.course_id, c.course_name;

select s.student_name, e.enrollment_id, e.fee_paid from enrollments e right join students s on 
e.student_id = s.student_id;

select student_name, city from students where city = 'Theni' union select student_name,city from students
where city = 'Chennai';

select course_name, course_fee from courses where course_fee > (select avg(course_fee) from courses);

select c.course_name, c.course_fee from courses c where c.course_id in (select e.course_id from enrollments e where e.score >= 90);

select s.student_name, c.course_name, c.course_fee, e.fee_paid from students s join enrollments e on 
s.student_id = e.student_id join courses c on e.course_id = c.course_id where e.fee_paid < c.course_fee;

select student_id, count(*) as total_courses, sum(fee_paid) as total_paid from enrollments group by student_id 
having count(*) > 1;

select c.course_name, avg(e.score) as avg_score from courses c join enrollments e on c.course_id = e.course_id 
group by c.course_id, c.course_name having avg(e.score) >= 70;

select enrollment_id, student_id, score, rank() over (order by score desc) as score_rank from 
enrollments where score is not null;

select course_id, score, dense_rank() over (partition by course_id order by score desc) as course_rank 
from enrollments where score is not null;

select student_id, enrollment_date, row_number() over (partition by student_id order by enrollment_date) 
as registration_no from enrollments;

select course_id, score, lag(score) over (partition by course_id order by enrollment_date) as previous_score 
from enrollments where score is not null;

select student_id, enrollment_date, lead(enrollment_date) over (partition by student_id order by enrollment_date) 
as next_enrollment_date from enrollments;

select course_id, score, first_value(score) over (partition by course_id order by score desc) as highest_score 
from enrollments where score is not null;

select enrollment_id, course_id, fee_paid, sum(fee_paid) over (partition by course_id order by enrollment_date)
as running_total from enrollments;

select enrollment_id, enrollment_status, case when enrollment_status = 'Completed' then 'Finished' when
 enrollment_status = 'Dropped' then 'Stopped' else 'In Progress' end as status_result from enrollments;

select course_name, course_fee, case when course_fee >= 10000 then 'High Fee' when course_fee >= 7000 then 
'Medium Fee' else 'Low Fee' end as fee_category from courses;

select c.course_name, c.category, count(e.enrollment_id) as total_enrollments, sum(e.fee_paid) as total_collection, avg(e.score)
 as avg_score from courses c join enrollments e on c.course_id = e.course_id group by c.course_id, c.course_name, c.category 
 having count(e.enrollment_id) >= 5 order by total_collection desc;
